const User = require("../models/User");
const Hostel = require("../models/Hostel");
const HostelMembership = require("../models/HostelMembership");
const Student = require("../models/Student");
const ROLES = require("../constants/roles");
const config = require("../config");
const { TokenValidationError, verifyAccessToken } = require("../utils/token");

const unauthorized = (res) => res.status(401).json({
    status: "error",
    message: "Authentication required"
});

const requireAuth = async (req, res, next) => {
    const authorization = req.get("authorization") || "";
    const match = /^Bearer (\S+)$/i.exec(authorization);
    if (!match) {
        return unauthorized(res);
    }

    let tokenPayload;
    try {
        tokenPayload = verifyAccessToken(match[1], config.jwtSecret);
    } catch (error) {
        if (error instanceof TokenValidationError) {
            return unauthorized(res);
        }
        return next(error);
    }

    try {
        const user = await User.findById(tokenPayload.sub)
            .select("name email mobile photoUrl role hostelId status isDeleted")
            .lean();

        if (!user || user.status !== "ACTIVE" || user.isDeleted || user.role !== tokenPayload.role) {
            return unauthorized(res);
        }

        let memberships = [];
        if (user.role !== ROLES.SYSTEM_ADMIN) {
            const membershipRecords = await HostelMembership.find({
                userId: user._id,
                role: user.role,
                status: "ACTIVE",
                isDeleted: { $ne: true }
            }).select("hostelId").lean();
            const memberHostelIds = [...new Set(membershipRecords.map((membership) => String(membership.hostelId)))];
            const activeHostels = memberHostelIds.length > 0
                ? await Hostel.find({
                    _id: { $in: memberHostelIds },
                    status: "ACTIVE",
                    isDeleted: { $ne: true }
                }).select("_id").lean()
                : [];
            memberships = activeHostels.map((hostel) => String(hostel._id));

            if (!tokenPayload.hostelId || !memberships.includes(tokenPayload.hostelId)) {
                return unauthorized(res);
            }
        } else if (tokenPayload.hostelId !== null) {
            const activeHostel = await Hostel.findOne({
                _id: tokenPayload.hostelId,
                status: "ACTIVE",
                isDeleted: { $ne: true }
            }).select("_id").lean();
            if (!activeHostel) {
                return unauthorized(res);
            }
        }

        let studentId = null;
        if (user.role === ROLES.STUDENT) {
            const student = await Student.findOne({
                userId: user._id,
                hostelId: tokenPayload.hostelId,
                status: "ACTIVE",
                isDeleted: { $ne: true }
            }).select("studentId").lean();

            if (!student) {
                return unauthorized(res);
            }
            studentId = student.studentId;
        }

        req.auth = {
            id: String(user._id),
            name: user.name,
            email: user.email || "",
            mobile: user.mobile || "",
            photoUrl: user.photoUrl || "",
            role: user.role,
            hostelId: tokenPayload.hostelId,
            memberships,
            studentId
        };
        return next();
    } catch (error) {
        if (error.name === "CastError") {
            return unauthorized(res);
        }
        return next(error);
    }
};

module.exports = requireAuth;

const Hostel = require("../models/Hostel");
const ROLES = require("../constants/roles");

const tenantForbidden = (res) => res.status(403).json({
    status: "error",
    message: "You do not have access to this hostel"
});

const tenantMiddleware = async (req, res, next) => {
    if (!req.auth?.role) {
        return res.status(401).json({
            status: "error",
            message: "Authentication required"
        });
    }

    const requestedHostelId = req.get("x-hostel-id");

    if (req.auth.role === ROLES.SYSTEM_ADMIN) {
        const selectedHostelId = requestedHostelId || req.auth.hostelId;
        if (!selectedHostelId) {
            req.tenant = null;
            return next();
        }

        try {
            const hostel = await Hostel.findOne({
                _id: selectedHostelId,
                status: "ACTIVE",
                isDeleted: { $ne: true }
            }).select("_id").lean();

            if (!hostel) {
                return tenantForbidden(res);
            }

            req.tenant = { hostelId: String(hostel._id) };
            return next();
        } catch (error) {
            if (error.name === "CastError") {
                return tenantForbidden(res);
            }
            return next(error);
        }
    }

    const memberships = req.auth.memberships || [];
    const selectedHostelId = requestedHostelId || req.auth.hostelId;

    if (!selectedHostelId || !memberships.includes(selectedHostelId)) {
        return tenantForbidden(res);
    }

    req.tenant = { hostelId: selectedHostelId };
    return next();
};

const requireTenant = (req, res, next) => {
    if (!req.tenant?.hostelId) {
        return res.status(403).json({
            status: "error",
            message: "A hostel context is required"
        });
    }

    return next();
};

module.exports = {
    requireTenant,
    tenantMiddleware
};

import { createContext, useCallback, useContext, useMemo, useState } from "react";
import { apiRequest } from "../services/api.js";

const AuthContext = createContext(null);
const SESSION_STORAGE_KEY = "shms.authSession";

const readStoredSession = () => {
  try {
    const stored = window.sessionStorage.getItem(SESSION_STORAGE_KEY);
    if (!stored) return null;
    const parsed = JSON.parse(stored);
    if (
      typeof parsed?.token === "string" &&
      parsed.token &&
      parsed.user &&
      typeof parsed.user.id === "string" &&
      typeof parsed.user.role === "string"
    ) {
      return parsed;
    }
  } catch {
    window.sessionStorage.removeItem(SESSION_STORAGE_KEY);
  }
  return null;
};

const storeSession = (session) => {
  try {
    if (session) {
      window.sessionStorage.setItem(SESSION_STORAGE_KEY, JSON.stringify(session));
    } else {
      window.sessionStorage.removeItem(SESSION_STORAGE_KEY);
    }
  } catch {
    window.sessionStorage.removeItem(SESSION_STORAGE_KEY);
  }
};

export function AuthProvider({ children }) {
  const [session, setSession] = useState(() => readStoredSession());

  const login = useCallback(async (credentials) => {
    const response = await apiRequest("/api/auth/login", {
      method: "POST",
      body: JSON.stringify(credentials)
    });
    setSession(response.data);
    storeSession(response.data);
    return response.data.user;
  }, []);

  const registerOwner = useCallback(async (profile) => {
    const response = await apiRequest("/api/auth/register", {
      method: "POST",
      body: JSON.stringify(profile)
    });
    setSession(response.data);
    storeSession(response.data);
    return response.data.user;
  }, []);

  const logout = useCallback(() => {
    setSession(null);
    storeSession(null);
  }, []);
  const updateUser = useCallback((user) => setSession((current) => {
    if (!current) return current;
    const next = { ...current, user: { ...current.user, ...user } };
    storeSession(next);
    return next;
  }), []);
  const value = useMemo(() => ({
    token: session?.token || null,
    user: session?.user || null,
    login,
    registerOwner,
    logout,
    updateUser
  }), [session, login, registerOwner, logout, updateUser]);

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth() {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider.");
  }
  return context;
}

export const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || "";

export class ApiError extends Error {
  constructor(message, status) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

export async function apiRequest(path, { token, signal, ...options } = {}) {
  const headers = new Headers(options.headers || {});

  if (options.body !== undefined && !(options.body instanceof FormData) && !headers.has("Content-Type")) {
    headers.set("Content-Type", "application/json");
  }

  if (token) {
    headers.set("Authorization", `Bearer ${token}`);
  }

  let response;
  try {
    response = await fetch(`${API_BASE_URL}${path}`, {
      ...options,
      headers,
      signal
    });
  } catch (error) {
    if (error.name === "AbortError") {
      throw error;
    }
    if (error instanceof TypeError) {
      throw new ApiError("Unable to connect to the server. Check your connection and try again.", 0);
    }
    throw error;
  }

  const payload = await response.json().catch(() => null);
  if (!response.ok) {
    throw new ApiError(payload?.message || "The request could not be completed.", response.status);
  }

  return payload;
}
